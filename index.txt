        // ========== PRECIOS (en DÓLARES) ==========
        // Insignia (logo pequeño) = $5
        // Pecho / espalda grande  = $10
        // Suéter                 = +$5 extra
        const PRECIO_INSIGNIA = 5;   // logo pequeño
        const PRECIO_GRANDE   = 10;  // pecho o espalda grande
        const EXTRA_SUETER    = 5;   // solo si es suéter

        // Umbral de escala: debajo = insignia, arriba = grande
        const UMBRAL_GRANDE = 90;    // escala >= 90% se considera "grande"

        function getSizeInfo(scale) {
            if (scale < UMBRAL_GRANDE) {
                return { texto: 'Insignia (pequeño)', color: '#22c55e', costoEstampa: PRECIO_INSIGNIA };
            }
            return { texto: 'Pecho / Espalda grande', color: '#f59e0b', costoEstampa: PRECIO_GRANDE };
        }

        function calcularPrecio() {
            const scale = parseFloat(document.getElementById('scale').value) || 100;
            const tieneDiseno = designImg.style.display !== 'none' && designImg.src && designImg.src.length > 10;

            let precio = 0;
            let detalle = '';

            if (!tieneDiseno) {
                document.getElementById('price-display').textContent = '$0';
                document.getElementById('size-badge').textContent = 'Tamaño: —';
                document.getElementById('size-badge').style.background = '#1e40af';
                document.getElementById('price-detail').textContent = 'Sube un diseño y ajusta el tamaño para ver el precio.';
                return 0;
            }

            const sizeInfo = getSizeInfo(scale);
            precio = sizeInfo.costoEstampa;

            // Extra por suéter SOLO si nosotros lo ponemos
            let extraSueter = 0;
            if (currentType === 'sueter' && duenoSueter === 'nuestro') {
                extraSueter = EXTRA_SUETER;
                precio += EXTRA_SUETER;
            }

            // Actualizar UI
            document.getElementById('price-display').textContent = '$' + precio;

            const badgeEl = document.getElementById('size-badge');
            badgeEl.textContent = sizeInfo.texto + ' (' + scale + '%)';
            badgeEl.style.background = sizeInfo.color;
            badgeEl.style.color = '#fff';

            const nombres = { camiseta: 'Camiseta', sueter: 'Suéter', gorra: 'Gorra' };
            let html = `Estampa: <b>${sizeInfo.texto}</b> → $${sizeInfo.costoEstampa}`;
            if (currentType === 'sueter') {
                if (extraSueter > 0) {
                    html += `<br>Suéter nuestro → +$${EXTRA_SUETER}`;
                } else {
                    html += `<br>Cliente trae suéter → +$0`;
                }
            }
            html += `<br>Producto: <b>${nombres[currentType] || currentType}</b>`;
            html += `<br><b>Total: $${precio}</b>`;
            document.getElementById('price-detail').innerHTML = html;

            return precio;
        }

            document.getElementById('btn-sueter-cliente').classList.toggle('active', quien === 'cliente');

        }

        function removeDesign() {
            designImg.src = '';

            posYRange.value = 0;
            posYNum.value = 0;

        }

        function removeBackground() {
            if (!designImg.src || designImg.style.display === 'none') {
                alert('Primero sube un diseño');
                return;
            }
            const img = new Image();
            img.crossOrigin = 'anonymous';
            img.onload = function() {
                const canvas = document.createElement('canvas');
                const ctx = canvas.getContext('2d');
                canvas.width = img.width;
                canvas.height = img.height;
                ctx.drawImage(img, 0, 0);
                const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
                const data = imageData.data;
                for (let i = 0; i < data.length; i += 4) {
                    const r = data[i], g = data[i+1], b = data[i+2];
                    if (r > 230 && g > 230 && b > 230) data[i+3] = 0;
                }
                ctx.putImageData(imageData, 0, 0);
                designImg.src = canvas.toDataURL('image/png');
            };
            img.src = designImg.src;
        }

        // Controles

        const posYRange = document.getElementById('posY');
        const posYNum = document.getElementById('posY-num');

        function updateTransform() {
            const s = scaleRange.value / 100;
            const x = posXRange.value;
            const y = posYRange.value;
            designImg.style.transform = `translate(${x}px, ${y}px) scale(${s})`;
            calcularPrecio();   // ← precio se actualiza al cambiar tamaño
        }

        function linkInputs(range, num) {
            range.addEventListener('input', () => {
                num.value = range.value;
                updateTransform();
            });
            num.addEventListener('input', () => {
                range.value = num.value;
                updateTransform();
            });
        }
        linkInputs(scaleRange, scaleNum);
        linkInputs(posXRange, posXNum);
        linkInputs(posYRange, posYNum);

        // Arrastrar diseño

        designImg.addEventListener('mousedown', e => {

        // Rotación 3D
        const viewport = document.getElementById('viewport');
        const mockup = document.getElementById('mockup');
        let isRotating = false;

            if (e.target === designImg || e.target.closest('.user-design')) return;

        });

        window.addEventListener('mousemove', e => {
            if (isDraggingDesign) {
                let nx = Math.max(-300, Math.min(300, initX + (e.clientX - startMX)));
                let ny = Math.max(-300, Math.min(300, initY + (e.clientY - startMY)));
                posXRange.value = nx;
                posXNum.value = nx;
                posYRange.value = ny;
                posYNum.value = ny;
                updateTransform();
                return;
            }
            if (isRotating) {
                const dx = e.clientX - rotStartX;
                const dy = e.clientY - rotStartY;
                rotY = startRotY + dx * 0.35;
                rotX = startRotX - dy * 0.35;
                rotX = Math.max(-28, Math.min(28, rotX));
                rotY = Math.max(-35, Math.min(35, rotY));
                mockup.style.transform = `rotateX(${rotX}deg) rotateY(${rotY}deg)`;
            }
        });

        window.addEventListener('mouseup', () => {
            isDraggingDesign = false;
            isRotating = false;
        });

        function reset3D() {
            rotX = -8;
            rotY = 0;
            mockup.style.transform = `rotateX(${rotX}deg) rotateY(${rotY}deg)`;
        }
        reset3D();

        // Zoom
        viewport.addEventListener('wheel', e => {
            if (designImg.style.display === 'none') return;
            e.preventDefault();
            let cur = parseFloat(scaleRange.value);
            cur += e.deltaY < 0 ? 8 : -8;
            cur = Math.max(5, Math.min(800, cur));
            scaleRange.value = cur;
            scaleNum.value = cur;
            updateTransform();
        }, { passive: false });

        // ========== GUARDAR DISEÑO ==========
        function saveDesign() {
            const shirtImg = document.getElementById('shirtImage');
            const designImgEl = document.getElementById('design');

            const canvas = document.createElement('canvas');
            const ctx = canvas.getContext('2d');
            
            canvas.width = 900;
            canvas.height = 950;

            ctx.fillStyle = '#f5f5f5';
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            function finishDownload() {
                try {
                    const link = document.createElement('a');
                    const fecha = new Date().toISOString().slice(0, 10);
                    const precio = calcularPrecio();
                    link.download = `Graphitech_${currentType}_${precio}pesos_${fecha}.png`;
                    link.href = canvas.toDataURL('image/png');
                    link.click();
                } catch (e) {
                    alert('El navegador bloqueó la descarga.\n\nSolución rápida:\n1. Presiona Windows + Shift + S\n2. Selecciona el área del diseño\n3. Pega en Paint y guarda');
                }
            }

            const shirt = new Image();
            shirt.src = shirtImg.src;

            shirt.onload = function() {
                const scale = Math.min(canvas.width / shirt.width, canvas.height / shirt.height) * 0.90;
                const w = shirt.width * scale;
                const h = shirt.height * scale;
                const x = (canvas.width - w) / 2;
                const y = (canvas.height - h) / 2;

                ctx.drawImage(shirt, x, y, w, h);

                if (designImgEl.src && designImgEl.style.display !== 'none') {
                    const design = new Image();
                    design.src = designImgEl.src;

                    design.onload = function() {
                        const s = parseFloat(document.getElementById('scale').value) / 100;
                        const posX = parseFloat(document.getElementById('posX').value);
                        const posY = parseFloat(document.getElementById('posY').value);

                        let printTop, printW, printH;
                        if (currentType === 'gorra') {
                            printTop = 0.28;
                            printW = 0.42;
                            printH = 0.28;
                        } else {
                            printTop = 0.26;
                            printW = 0.36;
                            printH = 0.45;
                        }

                        const areaX = x + w * ((1 - printW) / 2);
                        const areaY = y + h * printTop;
                        const areaWidth = w * printW;
                        const areaHeight = h * printH;

                        const designW = design.width * s * (areaWidth / 280);
                        const designH = design.height * s * (areaHeight / 280);

                        const dx = areaX + (areaWidth - designW) / 2 + (posX * 0.8);
                        const dy = areaY + (areaHeight - designH) / 2 + (posY * 0.8);

                        ctx.drawImage(design, dx, dy, designW, designH);
                        finishDownload();
                    };

                    design.onerror = function() {
                        finishDownload();
                    };
                } else {
                    finishDownload();
                }
            };

            shirt.onerror = function() {
                alert('No se pudo cargar la imagen de la prenda');
            };
        }

        // Precio inicial
        calcularPrecio();
    </script>
